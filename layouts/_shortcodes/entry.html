{{- /* 文档卡片：{{< entry "slug" >}}；slug 为 content/archive/ 下的文件名。 卡片按正文里出现的顺序排列；term.html 会读取本页引用过的 slug，末尾只补没被引用的条目。 */ -}} {{- $slug := .Get "slug" -}} {{- if not $slug }}{{ $slug = .Get 0 }}{{ end -}} {{- $slug = strings.TrimSuffix "/" (strings.TrimSpace $slug) -}} {{- $page := .Page.Site.GetPage (printf "/archive/%s" $slug) -}} {{- if not $page -}} {{- warnf "entry 短代码：找不到 /archive/%s（页面：%s）" $slug .Page.RelPermalink -}} ［缺少条目：{{ $slug }}］ {{- else -}} {{- $store := .Page.Store -}} {{- $used := $store.Get "entry-slugs" -}} {{- if not $used }}{{ $used = slice }}{{ end -}} {{- if not (in $used $slug) -}} {{- $store.Set "entry-slugs" ($used | append $slug) -}} {{- end -}} {{- $desc := $page.Description -}} {{ $page.Title }} {{- with $desc }} {{ . }} {{- end }} {{ $page.Date.Format "2006-01-02" }} → {{- end -}}
